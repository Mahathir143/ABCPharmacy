2.1 Domain Models

Data models utilize System.ComponentModel.DataAnnotations for declarative request validation:

// Models/Medicine.cs
public class Medicine
{
    public string Id { get; set; } = Guid.NewGuid().ToString("N");

    [Required(ErrorMessage = "Full Name of the medicine is required")]
    [StringLength(150, MinimumLength = 2)]
    public string FullName { get; set; } = string.Empty;

    public string? Notes { get; set; } = string.Empty;

    [Required(ErrorMessage = "Expiry Date is required")]
    public DateTime ExpiryDate { get; set; }

    public string Brand { get; set; } = string.Empty;
}

2.2 Thread-Safe Persistence with SemaphoreSlim

Because JSON flat files do not possess an underlying ACID storage engine, concurrent writes can lead to file-lock exceptions and data corruption. PharmacyService implements an asynchronous mutual exclusion lock using SemaphoreSlim(1, 1): 

public class PharmacyService : IPharmacyService
{
    private readonly SemaphoreSlim _semaphore = new(1, 1);

    public async Task<SaleRecord> RecordSaleAsync(CreateSaleDto dto)
    {
        await _semaphore.WaitAsync();
        try
        {
            var medicines = await ReadMedicinesInternalAsync();
            var medicine = medicines.FirstOrDefault(m => m.Id == dto.MedicineId)
                ?? throw new KeyNotFoundException($"Medicine '{dto.MedicineId}' not found.");

            if (medicine.Quantity < dto.Quantity)
                throw new InvalidOperationException($"Insufficient stock ({medicine.Quantity}).");

            // Deduct stock atomically
            medicine.Quantity -= dto.Quantity;
            await WriteMedicinesInternalAsync(medicines);

            // Record sale transaction
            var sale = new SaleRecord { ... };
            var sales = await ReadSalesInternalAsync();
            sales.Add(sale);
            await WriteSalesInternalAsync(sales);

            return sale;
        }
        finally
        {
            _semaphore.Release();
        }
    }
}

3. REST API Specification

Method Endpoint Description Status Codes
GET /api/medicines Returns all medicines. Supports optional ?search=query parameter. 200 OK
GET /api/medicines/{id} Returns single medicine record with complete details (including Notes). 200 OK, 404 Not Found
POST /api/medicines Creates a new medicine in inventory and persists to medicines.json. 201 Created, 400 Bad Request
PUT /api/medicines/{id} Updates an existing medicine details. 200 OK, 400 Bad Request, 404 Not Found
DELETE /api/medicines/{id} Removes medicine from inventory. 204 No Content, 404 Not Found
GET /api/sales Returns all sales records ordered by saleDate DESC. 200 OK
POST /api/sales Records sale, validates stock, decrements medicine quantity, persists sale record. 201 Created, 400 Bad Request, 404 Not Found

4. Frontend SPA Architecture (React 18)

4.1 State Management & React Hooks (wwwroot/js/app.js)

The single page application is built using modern React 18 functional components leveraging core hooks:

  • useState: Manages inventory list, sales records, search input, status filters, active modals, and form inputs.
  • useEffect: Triggers asynchronous API fetches on initial component mount (fetchMedicines, fetchSales).
  • useMemo: Memoizes computationally expensive derived state, including the filtered/sorted inventory list and live total sale calculations.

4.2 Real-time Search & Filter via useMemo

const filteredMedicines = useMemo(() => {
  let list = [...medicines];

  // 1. Search Query on Name or Brand
  if (searchQuery.trim()) {
    const q = searchQuery.toLowerCase().trim();
    list = list.filter(m =>
      m.fullName.toLowerCase().includes(q) ||
      m.brand.toLowerCase().includes(q)
    );
  }

  // 2. Status Filters
  if (statusFilter === 'expiring') {
    list = list.filter(m => isExpiringSoon(m));
  } else if (statusFilter === 'lowstock') {
    list = list.filter(m => isLowStock(m));
  } else if (statusFilter === 'both') {
    list = list.filter(m => isExpiringSoon(m) && isLowStock(m));
  } else if (statusFilter === 'normal') {
    list = list.filter(m => !isExpiringSoon(m) && !isLowStock(m));
  }

  // 3. Dynamic Multi-Column Sorting
  list.sort((a, b) => {
    let valA = a[sortBy], valB = b[sortBy];
    if (sortBy === 'expiry') {
      valA = new Date(a.expiryDate).getTime();
      valB = new Date(b.expiryDate).getTime();
    }
    if (valA < valB) return sortOrder === 'asc' ? -1 : 1;
    if (valA > valB) return sortOrder === 'asc' ? 1 : -1;
    return 0;
  });

  return list;
}, [medicines, searchQuery, statusFilter, sortBy, sortOrder]);

4.3 Color Indication Engine

// Calculates exact calendar days until expiration
const getDaysUntilExpiry = (dateStr) => {
  if (!dateStr) return 999;
  const today = new Date();
  today.setHours(0, 0, 0, 0);
  const expiry = new Date(dateStr);
  expiry.setHours(0, 0, 0, 0);
  return Math.ceil((expiry.getTime() - today.getTime()) / (1000 * 60 * 60 * 24));
};

// Requirement: Red background if expiry date < 30 days
const isExpiringSoon = (medicine) => getDaysUntilExpiry(medicine.expiryDate) < 30;

// Requirement: Yellow background if quantity in stock < 10
const isLowStock = (medicine) => medicine.quantity < 10;

// Row CSS class resolver
const getRowClass = (medicine) => {
  const expiring = isExpiringSoon(medicine);
  const lowStock = isLowStock(medicine);

  if (expiring && lowStock) return 'row-critical-both';
  if (expiring) return 'row-expiring-soon';
  if (lowStock) return 'row-low-stock';
  return 'row-normal';
};
